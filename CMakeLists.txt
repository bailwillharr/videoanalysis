cmake_minimum_required(VERSION 3.25)

project(videoanalysis)

set(OpenCV_ARCH "x64")
set(OpenCV_RUNTIME vc16)
if (WIN32)
    set(OpenCV_DIR "../opencv/build")
endif()
find_package(OpenCV REQUIRED)
if(WIN32)
    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        get_target_property(OpenCV_DLL opencv_world IMPORTED_LOCATION_DEBUG)
    else()
        get_target_property(OpenCV_DLL opencv_world IMPORTED_LOCATION_RELEASE)
    endif()
endif()
file(GLOB OpenCV_FFMPEG_DLL "${OpenCV_DIR}/bin/*.dll")

if (UNIX)
    include_directories(/usr/include/opencv4)
    link_libraries(${OpenCV_LIBRARIES})
endif()

add_subdirectory(simple_decoder)
add_subdirectory(text_decoder)
add_subdirectory(calibrate)
add_subdirectory(calibratetext)
add_subdirectory(text_decoder2)
add_subdirectory(ratetest)
add_subdirectory(ratetest_freq)
add_subdirectory(bayer_decode)
add_subdirectory(bayer_calib_extract)
add_subdirectory(color_correct_image)
